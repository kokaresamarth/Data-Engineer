2) Find duplicate records (e.g., duplicate names or emails) in a table.
=> employees

emp_id	name	dept_id
------  ----    -------
1	Asha	10
2	Ben	10
3	Asha	10
4	Ben	20

-> Find duplicates based on name and dept_id
--------------------------------------------
select name, dept_id, count(*) as duplicate
from employees
group by name, dept_id
having count(*) > 1;

=> OUTPUT
name	dept_id		duplicate
Asha	10		2

-> If duplicate names only then
-------------------------------
select name, count(*) as duplicate
from employees
group by name
having count(*) > 1;

=> OUTPUT 
name	duplicate
Asha	2
Ben	2

JOINS
=====
### Fetch the employee name and the department name they belongs to ###

1) Inner JOIN / JOIN 
-> it will try to fetch only those records which are prestent in both the tables

select e.emp_name, d.dept_name 
from employee e INNER JOIN department d ON e.dept_id = d.dept_id;

2) LEFT JOIN
### Fetch all the employee name and their department name they belongs to ###

select e.emp_name,  d.dept_name
from employee e Left Join department d ON e.dept_id = d.dept_id;

-- Left join = inner join + any additional records from the left table

3) RIGHT Join
### Fetch the employee name and their all department name they belongs to ###

select e.emp_name, d.dept_name
from employee e Right join department d ON e.dept_id = d.dept_id;
 
-- Right join = inner join + any additional records from the right table


--- Fetch details of ALL emp, their manager, their department and the projects they are work on. -----

select e.emp_name, d.dept_name, m.manager_name, p.project_name 
from employee e
LEFT JOIN department d ON  e.dept_id = d.dept_id
INNER JOIN manager m ON m.manager_id = e.manager_id
LEFT JOIN projects p ON p.emp_id = e.emp_id;

4) FULL JOIN

select e.emp_name, d.dept_name
from employee e 
FULL JOIN depatment d ON e.dept_name = d.dept_name;

--- full join = Inner join + all remaining records from Left table + all remaining record from Right table
