2) Find duplicate records (e.g., duplicate names or emails) in a table.
=> employees

emp_id	name	dept_id
------  ----    -------
1	Asha	10
2	Ben	10
3	Asha	10
4	Ben	20

-> Find duplicates based on name and dept_id
--------------------------------------------
select name, dept_id, count(*) as duplicate
from employees
group by name, dept_id
having count(*) > 1;

=> OUTPUT
name	dept_id		duplicate
Asha	10		2

-> If duplicate names only then
-------------------------------
select name, count(*) as duplicate
from employees
group by name
having count(*) > 1;

=> OUTPUT 
name	duplicate
Asha	2
Ben	2

--- Fetch the employee name and the department name they belongs to ---

-- Inner JOIN / JOIN 
-> it will try to fetch only those records which are prestent in both the tables

select e.emp_name, d.dept_name 
from employee e INNER JOIN department d ON e.dept_id = d.dept_id;
